  Future<Database> initDB() async {
    final path = kIsWeb
        ? '/bookstore_eproject.db'
        : join(await getDatabasesPath(), 'bookstore_eproject.db');
    return openDatabase(
      path,
      version: 3,
      onCreate: _onCreate,
      onUpgrade: _onUpgrade,
    );
  }

  Future<void> _onCreate(Database db, int version) async {
    await db.execute('''
      CREATE TABLE users (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        name TEXT, email TEXT UNIQUE, password TEXT, phone TEXT,
        address TEXT, paymentMethod TEXT, role TEXT
      )''');
    await db.execute('''
      CREATE TABLE books (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        title TEXT, author TEXT, genre TEXT, description TEXT, coverUrl TEXT,
        price REAL, popularity INTEGER, releaseDate TEXT,
        isBestseller INTEGER, isNewArrival INTEGER
      )''');
    await db.execute('''
      CREATE TABLE cart (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        userId INTEGER, bookId INTEGER, quantity INTEGER
      )''');

    await db.execute('''

      )''');
    await db.execute('''
      CREATE TABLE orders (
        id INTEGER PRIMARY KEY AUTOINCREMENT,

  Future<void> _onUpgrade(Database db, int oldVersion, int newVersion) async {
    if (oldVersion < 2) {
      try {
        await db.execute('ALTER TABLE orders ADD COLUMN itemsSummary TEXT');
      } catch (_) {}
      // Catalog was replaced with the bigger one, so old references are cleared.
      await db.delete('cart');
      await db.delete('wishlist');
      await db.delete('reviews');
      await db.delete('books');
      await _seedBooks(db);
    }
    if (oldVersion < 3) {
      for (final col in ['customerName', 'phone', 'note']) {
        try {
          await db.execute('ALTER TABLE orders ADD COLUMN $col TEXT');
        } catch (_) {}
      }
    }
  }

    final d = await db;

        whereArgs: [email.trim().toLowerCase(), phone.trim()]);
  }

  Future<int> updateProfile(int id, Map<String, dynamic> data) async {

  Future<void> deleteBook(int id) async {

      await txn.delete('wishlist', where: 'bookId = ?', whereArgs: [id]);
      await txn.delete('reviews', where: 'bookId = ?', whereArgs: [id]);
      await txn.delete('books', where: 'id = ?', whereArgs: [id]);
    });
  }

  // --- CART ---
  Future<void> addToCart(int userId, int bookId) async {
    final d = await db;
    final check = await d.query('cart',
        where: 'userId = ? AND bookId = ?', whereArgs: [userId, bookId]);
    if (check.isNotEmpty) {
      final qty = (check.first['quantity'] as num).toInt();
      await d.update('cart', {'quantity': qty + 1},
          where: 'id = ?', whereArgs: [check.first['id']]);
    } else {
      await d.insert('cart', {'userId': userId, 'bookId': bookId, 'quantity': 1});
    }
  }

  Future<Rows> getCartItems(int userId) async {
    final d = await db;
    return _rows(await d.rawQuery('''
      SELECT cart.id AS cartId, cart.quantity, books.*
      FROM cart INNER JOIN books ON cart.bookId = books.id
      WHERE cart.userId = ?
      ORDER BY cart.id DESC
    ''', [userId]));
  }

  Future<int> getCartCount(int userId) async {
    final d = await db;
    final r = await d.rawQuery(
        'SELECT COALESCE(SUM(quantity), 0) AS c FROM cart WHERE userId = ?', [userId]);
    return (r.first['c'] as num).toInt();
  }

  Future<void> updateCartQty(int cartId, int newQty) async {
    final d = await db;
    if (newQty <= 0) {
      await d.delete('cart', where: 'id = ?', whereArgs: [cartId]);
    } else {
      await d.update('cart', {'quantity': newQty}, where: 'id = ?', whereArgs: [cartId]);
    }
  }

  // --- WISHLIST ---
  /// Returns true when the book is in the wishlist after the call.
  Future<bool> toggleWishlist(int userId, int bookId) async {
    final d = await db;
    final check = await d.query('wishlist',
        where: 'userId = ? AND bookId = ?', whereArgs: [userId, bookId]);
    if (check.isNotEmpty) {
      await d.delete('wishlist', where: 'id = ?', whereArgs: [check.first['id']]);
      return false;
    }
    await d.insert('wishlist', {'userId': userId, 'bookId': bookId});
    return true;
  }

  Future<bool> isInWishlist(int userId, int bookId) async {
    final d = await db;
    final r = await d.query('wishlist',
        where: 'userId = ? AND bookId = ?', whereArgs: [userId, bookId], limit: 1);
    return r.isNotEmpty;
  }

  Future<Set<int>> getWishlistIds(int userId) async {
    final d = await db;
    final r = await d.query('wishlist',
        columns: ['bookId'], where: 'userId = ?', whereArgs: [userId]);
    return r.map((e) => (e['bookId'] as num).toInt()).toSet();
  }

  Future<Rows> getWishlist(int userId) async {
    final d = await db;
    return _rows(await d.rawQuery('''
      SELECT wishlist.id AS wishId, books.*
      FROM wishlist INNER JOIN books ON wishlist.bookId = books.id
      WHERE wishlist.userId = ?
      ORDER BY wishlist.id DESC
    ''', [userId]));
  }

  Future<Rows> getReviews(int bookId) async {
    final d = await db;
    return _rows(await d.query('reviews',

  }

  /// bookId -> {avg: double, count: int}
  Future<Map<int, Map<String, dynamic>>> getRatingsMap() async {
    final d = await db;
    final r = await d.rawQuery(
        'SELECT bookId, AVG(rating) AS avg, COUNT(*) AS cnt FROM reviews GROUP BY bookId');
    return {
      for (final row in r)
        (row['bookId'] as num).toInt(): {
          'avg': (row['avg'] as num).toDouble(),
          'count': (row['cnt'] as num).toInt(),
        }
    };
  }

  Future<int> addReview(Map<String, dynamic> review) async {
    final d = await db;
    return d.insert('reviews', review);
  }

  Future<void> likeReview(int reviewId) async {
    final d = await db;
    await d.rawUpdate('UPDATE reviews SET likes = COALESCE(likes, 0) + 1 WHERE id = ?', [reviewId]);
  }

  // --- ORDERS ---
  /// Places the order and empties the cart in one transaction.
  Future<int> checkout(int userId, Map<String, dynamic> order) async {
    final d = await db;
    return d.transaction((txn) async {
      final id = await txn.insert('orders', order);
      await txn.delete('cart', where: 'userId = ?', whereArgs: [userId]);
      return id;
    });
  }

  Future<Rows> getUserOrders(int userId) async {
    final d = await db;
    return _rows(await d.query('orders',
        where: 'userId = ?', whereArgs: [userId], orderBy: 'id DESC'));
  }

  Future<Rows> getAllOrders() async {
    final d = await db;
    return _rows(await d.rawQuery('''
      SELECT orders.*, COALESCE(NULLIF(orders.customerName, ''), users.name) AS buyerName
      FROM orders LEFT JOIN users ON users.id = orders.userId
      ORDER BY orders.id DESC
    '''));
  }

  Future<void> updateOrderStatus(int orderId, String status) async {
    final d = await db;
    await d.update('orders', {'status': status}, where: 'id = ?', whereArgs: [orderId]);
  }

  Future<void> loadData() async {
    final b = await dbHelper.getBooks();
    final o = await dbHelper.getAllOrders();
    if (!mounted) return;
    setState(() {
      books = b;
      orders = o;
      loading = false;
    });

  void _toast(String m) {
    if (!mounted) return;
    ScaffoldMessenger.of(context)
      ..hideCurrentSnackBar()
      ..showSnackBar(SnackBar(content: Text(m)));
  }

  Future<void> _openForm([Map<String, dynamic>? book]) async {
    final result = await showModalBottomSheet<Map<String, dynamic>>(
      context: context,
      isScrollControlled: true,
      backgroundColor: Colors.white,
      shape: const RoundedRectangleBorder(
          borderRadius: BorderRadius.vertical(top: Radius.circular(20))),
      builder: (ctx) => Padding(
        padding: EdgeInsets.only(bottom: MediaQuery.of(ctx).viewInsets.bottom),
        child: _BookForm(book: book),
      ),
    );
    if (result == null) return;
    try {
      if (book == null) {
        await dbHelper.addBook(result);
        _toast('Book added');
      } else {
        await dbHelper.updateBook(asInt(book['id']), result);
        _toast('Book updated');
      }
      await loadData();
    } catch (e) {
      _toast('Could not save the book: $e');
    }
  }

    final ok = await showDialog<bool>(

      builder: (ctx) => AlertDialog(
        title: const Text('Delete this book?'),

        actions: [
          TextButton(onPressed: () => Navigator.pop(ctx, false), child: const Text('Keep')),

            child: const Text('Delete'),

/// Asks for a numeric PIN. Pops true when the PIN is correct.
class _PinDialog extends StatefulWidget {
  final String title;
  final String expectedPin;
  const _PinDialog({required this.title, required this.expectedPin});

  @override
  State<_PinDialog> createState() => _PinDialogState();
}

class _PinDialogState extends State<_PinDialog> {
  final TextEditingController ctrl = TextEditingController();
  String? error;
  int attempts = 0;

  @override
  void dispose() {
    ctrl.dispose();
    super.dispose();
  }

  void _check() {
    if (ctrl.text.trim() == widget.expectedPin) {
      Navigator.pop(context, true);
      return;
    }
    setState(() {
      attempts++;
      error = attempts >= 3 ? 'Incorrect PIN. Please try again later.' : 'Incorrect PIN';
      ctrl.clear();
    });
    if (attempts >= 3) {
      Future.delayed(const Duration(milliseconds: 900), () {
        if (mounted) Navigator.pop(context, false);
      });
    }
  }

  @override
  Widget build(BuildContext context) {
    return AlertDialog(
      icon: const Icon(Icons.lock_outline, color: AppColors.teal),
      title: Text(widget.title),
      content: TextField(
        controller: ctrl,
        autofocus: true,
        obscureText: true,
        keyboardType: TextInputType.number,
        maxLength: 8,
        enabled: attempts < 3,
        textAlign: TextAlign.center,
        style: const TextStyle(fontSize: 22, letterSpacing: 6),
        decoration: InputDecoration(counterText: '', hintText: 'Enter PIN', errorText: error),
        onSubmitted: (_) => _check(),
      ),
      actions: [
        TextButton(onPressed: () => Navigator.pop(context, false), child: const Text('Cancel')),
        FilledButton(onPressed: attempts < 3 ? _check : null, child: const Text('Continue')),
      ],
    );
  }
}

  void _applyFilters() {
    final q = query.toLowerCase();
    final list = allBooks.where((b) {
      final matchesSearch = q.isEmpty ||
          b['title'].toString().toLowerCase().contains(q) ||
          b['author'].toString().toLowerCase().contains(q) ||
          b['genre'].toString().toLowerCase().contains(q);
      final matchesGenre = genre == 'All' || b['genre'] == genre;
      final matchesSpecial = special == 'All' ||
          (special == 'Bestsellers' && asInt(b['isBestseller']) == 1) ||
          (special == 'New Arrivals' && asInt(b['isNewArrival']) == 1);
      return matchesSearch && matchesGenre && matchesSpecial;
    }).toList();

    int cmp(Map<String, dynamic> a, Map<String, dynamic> b) {
      switch (sortBy) {
        case 'Price: Low to High':
          return asDouble(a['price']).compareTo(asDouble(b['price']));
        case 'Price: High to Low':
          return asDouble(b['price']).compareTo(asDouble(a['price']));
        case 'Newest':
          return b['releaseDate'].toString().compareTo(a['releaseDate'].toString());
        default:
          return asInt(b['popularity']).compareTo(asInt(a['popularity']));
      }
    }

    list.sort(cmp);
    filtered = list;
  }

  final s = v.round().toString();

class _LogoPainter extends CustomPainter {
  final bool light;
  _LogoPainter(this.light);

  @override
  void paint(Canvas canvas, Size size) {
    final s = size.width;
    final tile = RRect.fromRectAndRadius(
        Rect.fromLTWH(0, 0, s, s), Radius.circular(s * 0.26));

    final bg = Paint();
    if (light) {
      bg.color = Colors.white;
    } else {
      bg.shader = const LinearGradient(
              begin: Alignment.topLeft,
              end: Alignment.bottomRight,
              colors: [AppColors.teal, AppColors.tealDark])
          .createShader(Rect.fromLTWH(0, 0, s, s));
    }
    canvas.drawRRect(tile, bg);

    final pageColor = light ? AppColors.teal : Colors.white;
    final lineColor = light ? Colors.white : AppColors.teal;
    final pages = Paint()..color = pageColor;

    Path left() => Path()
      ..moveTo(0.485 * s, 0.37 * s)
      ..quadraticBezierTo(0.32 * s, 0.30 * s, 0.17 * s, 0.36 * s)
      ..lineTo(0.17 * s, 0.75 * s)
      ..quadraticBezierTo(0.32 * s, 0.69 * s, 0.485 * s, 0.78 * s)
      ..close();
    Path right() => Path()
      ..moveTo(0.515 * s, 0.37 * s)
      ..quadraticBezierTo(0.68 * s, 0.30 * s, 0.83 * s, 0.36 * s)
      ..lineTo(0.83 * s, 0.75 * s)
      ..quadraticBezierTo(0.68 * s, 0.69 * s, 0.515 * s, 0.78 * s)
      ..close();
    canvas.drawPath(left(), pages);
    canvas.drawPath(right(), pages);

    final line = Paint()
      ..color = tint(lineColor, 0.4)
      ..strokeWidth = s * 0.022
      ..strokeCap = StrokeCap.round;
    for (final y in [0.49, 0.59]) {
      canvas.drawLine(Offset(0.25 * s, (y + 0.015) * s), Offset(0.43 * s, (y + 0.035) * s), line);
      canvas.drawLine(Offset(0.57 * s, (y + 0.035) * s), Offset(0.75 * s, (y + 0.015) * s), line);
    }

    final roof = Paint()
      ..color = AppColors.marigold
      ..style = PaintingStyle.stroke
      ..strokeWidth = s * 0.075
      ..strokeCap = StrokeCap.round
      ..strokeJoin = StrokeJoin.round;
    final roofPath = Path()
      ..moveTo(0.20 * s, 0.30 * s)
      ..lineTo(0.5 * s, 0.12 * s)
      ..lineTo(0.80 * s, 0.30 * s);
    canvas.drawPath(roofPath, roof);
  }

  @override
  bool shouldRepaint(covariant _LogoPainter old) => old.light != light;

/// Shows the cover image. If it is missing or fails to load (offline, CORS,
/// bad link) a designed gradient cover with the title is shown instead.
class BookCover extends StatelessWidget {
  final Map<String, dynamic> book;
  final BoxFit fit;
  const BookCover({super.key, required this.book, this.fit = BoxFit.cover});

  static const List<List<Color>> _palettes = [
    [Color(0xFF0E6B63), Color(0xFF083F3B)],
    [Color(0xFF3B4A8C), Color(0xFF1F2756)],
    [Color(0xFF9C4A35), Color(0xFF5E2517)],
    [Color(0xFF6B4E8E), Color(0xFF392A55)],
    [Color(0xFF2F6F9E), Color(0xFF173B58)],
    [Color(0xFF8A6D1F), Color(0xFF4D3B0B)],
  ];

  @override
  Widget build(BuildContext context) {
    final url = (book['coverUrl'] ?? '').toString().trim();
    if (url.isEmpty) return _fallback();
    return Image.network(
      url,
      fit: fit,
      cacheWidth: 400,
      filterQuality: FilterQuality.medium,
      gaplessPlayback: true,
      loadingBuilder: (ctx, child, progress) => progress == null ? child : _fallback(),
      errorBuilder: (ctx, error, stack) => _fallback(),
    );
  }

  Widget _fallback() {
    final title = (book['title'] ?? '').toString();
    final author = (book['author'] ?? '').toString();
    final seed = title.codeUnits.fold<int>(0, (a, b) => a + b);
    final colors = _palettes[seed % _palettes.length];
    return LayoutBuilder(builder: (ctx, c) {
      final w = c.maxWidth.isFinite ? c.maxWidth : 120.0;
      return Container(
        decoration: BoxDecoration(
          gradient: LinearGradient(
              begin: Alignment.topLeft, end: Alignment.bottomRight, colors: colors),
        ),
        child: Row(
          children: [
            Container(width: (w * 0.06).clamp(4.0, 12.0), color: tint(Colors.black, 0.2)),
            Expanded(
              child: Padding(
                padding: EdgeInsets.all((w * 0.09).clamp(8.0, 18.0)),
                child: Column(
                  mainAxisAlignment: MainAxisAlignment.spaceBetween,
                  crossAxisAlignment: CrossAxisAlignment.start,
                  children: [
                    Text(title,
                        maxLines: 4,
                        overflow: TextOverflow.ellipsis,
                        style: serif((w / 8.5).clamp(11.0, 26.0), color: Colors.white)),
                    Text(author,
                        maxLines: 2,
                        overflow: TextOverflow.ellipsis,
                        style: TextStyle(
                            color: tint(Colors.white, 0.75),
                            fontSize: (w / 12).clamp(9.0, 14.0))),
                  ],
                ),
              ),
            ),
          ],
        ),
      );
    });
  }
}
